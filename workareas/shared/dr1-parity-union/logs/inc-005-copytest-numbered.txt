19:const leaves = (node, path = []) =>
20:  typeof node === 'object' && node !== null
21:    ? Object.entries(node).flatMap(([key, value]) =>
22:        leaves(value, [...path, key])
23:      )
24:    : [{ path: path.join('.'), value: node }]

26:describe('#copy', () => {
27:  // Parameterised strings are copy FUNCTIONS: a leaf may be a function of
28:  // sample arguments returning the finished sentence.
29:  it('Should have a non-empty string (or string-returning function) at every leaf', () => {
30:    for (const { path, value } of leaves(copy)) {
31:      const text =
32:        typeof value === 'function' ? value('sample', 'sample') : value
33:      expect(typeof text, `${path} must resolve to a string`).toBe('string')
34:      expect(text.trim().length, `${path} must not be empty`).toBeGreaterThan(0)
35:    }
